# Slack Message/Thread as Task Context — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Let users attach Slack messages (single or thread) to a Lumo task as durable, refreshable context — visible in a new Task Detail tab and inlined into the `lumo task context` markdown the agent reads.

**Architecture:** New `TaskSlackContext` Prisma model with a JSON snapshot column (one row per "add" action). A dedicated server-only fetcher hits Slack's Web API via raw `fetch` (reusing the pattern in `notify.ts`). Strict layer separation: route → service → repository → Prisma. UI mirrors the Activity / Agent tab pattern. Server pre-renders the markdown section so the CLI just splices a string.

**Tech Stack:** Next.js App Router, TypeScript (strict), Prisma + Neon Postgres, Tailwind + shadcn/ui, Jest (jsdom + node), react-intl.

**Spec:** `docs/superpowers/specs/2026-05-19-slack-task-context-design.md`

**Branch:** `lumo/LUM-44-add-slack-task-context` (already created from `origin/main`)

## Conventions used throughout

- **Imports**: `@/lib/...` alias (Next path mapping). Tests use the same alias.
- **Repository shape**: `export const xRepository = { method() {} }` — see `lib/repositories/session.repository.ts`.
- **Service errors**: `throw Object.assign(new Error('error_code'), { status: <http_code> })`. Routes catch via `'status' in error`.
- **Route auth**: `requireAuth()` then load task with `taskRepository.findWithProjectWorkspaceId(id)`, then `requireWorkspaceMember(workspaceId, userId)`. Returns `memberResult.id` for the actor.
- **Tests**: node-env tests prefix with `/** @jest-environment node */` (default is jsdom). Prisma mocked via `jest.mock('@/lib/db/prisma', ...)`.
- **Commits**: One per task at the end. Include `[LUM-44]` suffix. Use `feat:` / `chore:` / `test:` prefixes consistent with recent history.

```ts
// lib/integrations/slack/parse-permalink.ts
export interface ParsedSlackPermalink {
  teamSubdomain: string
  channelId: string
  ts: string
  threadTs: string | undefined
}

const URL_REGEX =
  /^https:\/\/([a-z0-9-]+)\.slack\.com\/archives\/(C[A-Z0-9]+)\/p(\d{16,})(\?.*)?$/

/**
 * Parses a Slack permalink. Returns null when the URL is not a recognized
 * archive-style message permalink. Recognizes both single-message and
 * thread-reply formats. The presence of a `thread_ts` query parameter is
 * what makes a permalink a reply, even if the message id itself looks
 * normal — Slack does not encode "this is a reply" anywhere else.
 */
export function parseSlackPermalink(
  input: string
): ParsedSlackPermalink | null {
  const match = URL_REGEX.exec(input.trim())
  if (!match) return null

  const [, teamSubdomain, channelId, pDigits, query] = match
  if (pDigits.length < 16) return null

  // p1700000000000100 -> 1700000000.000100 (insert "." 6 chars from the right)
  const ts = pDigits.slice(0, pDigits.length - 6) + '.' + pDigits.slice(-6)

  let threadTs: string | undefined
  if (query) {
    const params = new URLSearchParams(query.slice(1))
    threadTs = params.get('thread_ts') ?? undefined
  }

  return { teamSubdomain, channelId, ts, threadTs }
}
```

```ts
// lib/integrations/slack/token.ts
import { prisma } from '@/lib/db/prisma'
import { decrypt } from '@/lib/crypto'

/**
 * Fetches the workspace's Slack bot token and decrypts it. Returns null
 * when the workspace hasn't installed the Slack app — callers should
 * surface a "Slack not connected" error.
 */
export async function getSlackIntegrationToken(
  workspaceId: string
): Promise<{ token: string; slackTeamId: string } | null> {
  const integration = await prisma.slackIntegration.findUnique({
    where: { workspaceId },
    select: { accessToken: true, slackTeamId: true },
  })
  if (!integration) return null
  return {
    token: decrypt(integration.accessToken),
    slackTeamId: integration.slackTeamId,
  }
}
```

```ts
// lib/integrations/slack/context-fetcher.ts
import type {
  SlackContextSnapshot,
  SlackContextMessage,
} from '@/types/slack-context'

export class SlackApiError extends Error {
  constructor(
    public readonly code: string,
    message?: string
  ) {
    super(message ?? code)
    this.name = 'SlackApiError'
  }
}

export interface RawSlackMessage {
  ts: string
  user: string
  text: string
  edited?: unknown
  files?: Array<{ name: string; permalink: string; mimetype: string }>
}

async function slackPost(
  token: string,
  method: string,
  body: object
): Promise<{ [k: string]: unknown }> {
  const res = await fetch(`https://slack.com/api/${method}`, {
    method: 'POST',
    headers: {
      Authorization: `Bearer ${token}`,
      'Content-Type': 'application/json',
    },
    body: JSON.stringify(body),
  })
  if (!res.ok) {
    throw new SlackApiError('slack_unavailable', `HTTP ${res.status}`)
  }
  const json = (await res.json()) as { ok: boolean; error?: string }
  if (!json.ok) {
    throw new SlackApiError(json.error ?? 'slack_unknown_error')
  }
  return json as { [k: string]: unknown }
}

export async function fetchSingleMessage(
  token: string,
  channelId: string,
  ts: string
): Promise<RawSlackMessage> {
  const json = await slackPost(token, 'conversations.history', {
    channel: channelId,
    latest: ts,
    inclusive: true,
    limit: 1,
  })
  const messages = (json.messages as RawSlackMessage[] | undefined) ?? []
  const msg = messages[0]
  if (!msg || msg.ts !== ts) {
    throw new SlackApiError('message_not_found')
  }
  return msg
}

export async function fetchThread(
  token: string,
  channelId: string,
  threadTs: string
): Promise<RawSlackMessage[]> {
  const all: RawSlackMessage[] = []
  let cursor: string | undefined
  while (true) {
    const json = await slackPost(token, 'conversations.replies', {
      channel: channelId,
      ts: threadTs,
      ...(cursor ? { cursor } : {}),
    })
    const batch = (json.messages as RawSlackMessage[] | undefined) ?? []
    all.push(...batch)
    const meta = json.response_metadata as { next_cursor?: string } | undefined
    if (json.has_more && meta?.next_cursor) {
      cursor = meta.next_cursor
    } else {
      break
    }
  }
  if (all.length === 0) throw new SlackApiError('message_not_found')
  return all
}

export async function resolveUsers(
  token: string,
  userIds: string[]
): Promise<Map<string, { name: string | null; avatarUrl: string | null }>> {
  const unique = Array.from(new Set(userIds))
  const out = new Map<
    string,
    { name: string | null; avatarUrl: string | null }
  >()
  await Promise.all(
    unique.map(async id => {
      try {
        const json = await slackPost(token, 'users.info', { user: id })
        const user = json.user as
          | {
              profile?: {
                display_name?: string
                real_name?: string
                image_72?: string
              }
            }
          | undefined
        const profile = user?.profile
        out.set(id, {
          name:
            profile?.display_name?.trim() || profile?.real_name?.trim() || null,
          avatarUrl: profile?.image_72 ?? null,
        })
      } catch {
        out.set(id, { name: null, avatarUrl: null })
      }
    })
  )
  return out
}

export async function resolveChannelName(
  token: string,
  channelId: string
): Promise<string | null> {
  try {
    const json = await slackPost(token, 'conversations.info', {
      channel: channelId,
    })
    const channel = json.channel as { name?: string } | undefined
    return channel?.name ?? null
  } catch {
    return null
  }
}

export function toSnapshot(
  rawMessages: RawSlackMessage[],
  users: Map<string, { name: string | null; avatarUrl: string | null }>
): SlackContextSnapshot {
  const messages: SlackContextMessage[] = rawMessages.map(m => {
    const u = users.get(m.user) ?? { name: null, avatarUrl: null }
    const out: SlackContextMessage = {
      ts: m.ts,
      userId: m.user,
      userName: u.name,
      userAvatarUrl: u.avatarUrl,
      text: m.text ?? '',
      edited: m.edited != null,
    }
    if (m.files && m.files.length > 0) {
      out.files = m.files.map(f => ({
        name: f.name,
        permalink: f.permalink,
        mimeType: f.mimetype,
      }))
    }
    return out
  })
  return { messages }
}
```

```ts
import { prisma } from '@/lib/db/prisma'
import type { Prisma } from '@prisma/client'
import type { SlackContextSnapshot } from '@/types/slack-context'

export interface CreateTaskSlackContextInput {
  taskId: string
  workspaceId: string
  permalink: string
  slackTeamId: string
  slackChannelId: string
  slackChannelName: string | null
  rootTs: string
  threadTs: string | null
  isThread: boolean
  snapshot: SlackContextSnapshot
  snapshotAt: Date
  createdById: string
}

export const taskSlackContextRepository = {
  async create(input: CreateTaskSlackContextInput) {
    return prisma.taskSlackContext.create({
      data: input as Prisma.TaskSlackContextUncheckedCreateInput,
    })
  },

  async listByTask(taskId: string) {
    return prisma.taskSlackContext.findMany({
      where: { taskId },
      orderBy: { createdAt: 'desc' },
    })
  },

  async getById(id: string) {
    return prisma.taskSlackContext.findUnique({ where: { id } })
  },

  /**
   * Two modes:
   * - Success: pass the new snapshot, clears refreshError, updates snapshotAt.
   * - Failure: pass null + error string; preserves the existing snapshot/snapshotAt
   *   so a refresh failure never destroys the cached content.
   */
  async updateSnapshot(
    id: string,
    snapshot: SlackContextSnapshot | null,
    refreshError: string | null
  ) {
    const data: Prisma.TaskSlackContextUpdateInput = snapshot
      ? {
          snapshot: snapshot as unknown as Prisma.InputJsonValue,
          snapshotAt: new Date(),
          refreshError,
        }
      : { refreshError }
    return prisma.taskSlackContext.update({ where: { id }, data })
  },

  async delete(id: string) {
    return prisma.taskSlackContext.delete({ where: { id } })
  },
}
```

```ts
import { z } from 'zod'

/**
 * Permission to match a real Slack archive permalink. We deliberately match
 * the full structure (including the C-prefixed channel id and a 16+ digit
 * `p…` segment) rather than just "starts with https" — so the route layer
 * never hands a clearly-bogus string to the service.
 */
export const slackPermalinkSchema = z
  .string()
  .regex(
    /^https:\/\/[a-z0-9-]+\.slack\.com\/archives\/C[A-Z0-9]+\/p\d{16,}(\?.*)?$/,
    'slackContext.error.invalidPermalink'
  )

export const linkSlackContextBodySchema = z.object({
  permalink: slackPermalinkSchema,
})
```

```ts
import { getSlackIntegrationToken } from '@/lib/integrations/slack/token'
import {
  fetchSingleMessage,
  fetchThread,
  resolveUsers,
  resolveChannelName,
  toSnapshot,
  SlackApiError,
} from '@/lib/integrations/slack/context-fetcher'
import { taskSlackContextRepository } from '@/lib/repositories/task-slack-context.repository'

function err(message: string, status: number): Error & { status: number } {
  return Object.assign(new Error(message), { status })
}

function mapSlackError(e: unknown): Error & { status: number } {
  if (e instanceof SlackApiError) {
    switch (e.code) {
      case 'not_in_channel':
        return err('not_in_channel', 403)
      case 'channel_not_found':
      case 'message_not_found':
        return err('message_gone', 404)
      case 'slack_unavailable':
        return err('slack_unavailable', 502)
      default:
        return err(`slack_error:${e.code}`, 502)
    }
  }
  return err('internal_error', 500)
}

export interface LinkSlackPermalinkInput {
  taskId: string
  workspaceId: string
  permalink: string
  actorMemberId: string
}

export const taskSlackContextService = {
  async listForTask(taskId: string) {
    return taskSlackContextRepository.listByTask(taskId)
  },

  async linkSlackPermalink(input: LinkSlackPermalinkInput) {
    const parsed = parseSlackPermalink(input.permalink)
    if (!parsed) throw err('invalid_permalink', 400)

    const integration = await getSlackIntegrationToken(input.workspaceId)
    if (!integration) throw err('not_connected', 412)

    if (parsed.teamSubdomain !== integration.slackTeamName) {
      throw err('wrong_team', 400)
    }

    let raw
    try {
      raw = parsed.threadTs
        ? await fetchThread(
            integration.token,
            parsed.channelId,
            parsed.threadTs
          )
        : [
            await fetchSingleMessage(
              integration.token,
              parsed.channelId,
              parsed.ts
            ),
          ]
    } catch (e) {
      throw mapSlackError(e)
    }

    const channelName = await resolveChannelName(
      integration.token,
      parsed.channelId
    )
    const users = await resolveUsers(
      integration.token,
      raw.map(m => m.user)
    )
    const snapshot = toSnapshot(raw, users)

    return taskSlackContextRepository.create({
      taskId: input.taskId,
      workspaceId: input.workspaceId,
      permalink: input.permalink,
      slackTeamId: integration.slackTeamId,
      slackChannelId: parsed.channelId,
      slackChannelName: channelName,
      rootTs: parsed.threadTs ?? parsed.ts,
      threadTs: parsed.threadTs ?? null,
      isThread: parsed.threadTs != null,
      snapshot,
      snapshotAt: new Date(),
      createdById: input.actorMemberId,
    })
  },
}
```

```ts
import { requireAuth, requireWorkspaceMember } from '@/lib/auth'
```

```ts
export async function GET(_request: NextRequest, { params }: Params) {
  try {
    const authResult = await requireAuth()
    if (authResult instanceof NextResponse) return authResult
    const { userId } = authResult
    const { id } = await params

    const task = await taskRepository.findWithProjectWorkspaceId(id)
    if (!task) return NextResponse.json({ error: 'Not found' }, { status: 404 })

    const memberResult = await requireWorkspaceMember(
      task.project.workspaceId,
      userId
    )
    if (memberResult instanceof NextResponse) return memberResult

    const contexts = await taskSlackContextService.listForTask(id)
    return NextResponse.json({ contexts })
  } catch (error) {
    if (error instanceof Error && 'status' in error) {
      return NextResponse.json(
        { error: error.message },
        { status: (error as Error & { status: number }).status }
      )
    }
    console.error('GET /api/tasks/[id]/slack-contexts error:', error)
    return NextResponse.json(
      { error: 'Internal Server Error' },
      { status: 500 }
    )

    const authResult = await requireAuth()
    if (authResult instanceof NextResponse) return authResult
    const { userId } = authResult
```

```ts
      setContexts(prev => [context, ...prev])
    },
    [taskId]
  )

    async (contextId: string) => {
      const { context } = await slackContextsClient.refresh(taskId, contextId)
      setContexts(prev => prev.map(c => (c.id === contextId ? context : c)))
    },
```

```tsx
// app/workspace/[workspaceUrl]/components/TaskDetailTimeline/slackMrkdwn.tsx
import React from 'react'

/**
 * Render a small subset of Slack mrkdwn into React nodes. Handles:
 *   *bold* → <strong>
 *   _italic_ → <em>
 *   <@U123> → @<userId or resolved name from map>
 *   <#C123|name> → #name
 *   <http://example.com|label> → <a>label</a>
 * Other mrkdwn (block quotes, lists) is rendered as plain text.
 */
export function SlackMrkdwn({
  text,
  userMap,
}: {
  text: string
  userMap?: Record<string, string | null>
}): React.ReactElement {
  const parts: Array<string | React.ReactElement> = []
  let i = 0
  let key = 0

  while (i < text.length) {
    const open = text.indexOf('<', i)
    if (open === -1) {
      parts.push(text.slice(i))
      break
    }
    if (open > i) parts.push(text.slice(i, open))
    const close = text.indexOf('>', open + 1)
    if (close === -1) {
      parts.push(text.slice(open))
      break
    }
    const token = text.slice(open + 1, close)
    if (token.startsWith('@') && /^@[UW][A-Z0-9]+$/.test(token)) {
      const uid = token.slice(1)
      const name = userMap?.[uid] ?? null
      parts.push(
        <span key={key++} className='text-blue-500'>
          @{name ?? uid}
        </span>
      )
    } else if (token.startsWith('#')) {
      const pipe = token.indexOf('|')
      const name = pipe >= 0 ? token.slice(pipe + 1) : 'channel'
      parts.push(
        <span key={key++} className='text-blue-500'>
          #{name}
        </span>
      )
    } else if (/^https?:\/\//.test(token)) {
      const pipe = token.indexOf('|')
      const url = pipe >= 0 ? token.slice(0, pipe) : token
      const label = pipe >= 0 ? token.slice(pipe + 1) : url
      parts.push(
        <a
          key={key++}
          href={url}
          target='_blank'
          rel='noopener noreferrer'
          className='text-blue-500 underline'
        >
          {label}
        </a>
      )
    } else {
      parts.push(`<${token}>`)
    }
    i = close + 1
  }

  // Now apply bold/italic to plain-string parts. Keep elements as-is.
  const rendered = parts.flatMap((p, idx) =>
    typeof p === 'string' ? applyEmphasis(p, idx, key) : [p]
  )

  return <>{rendered}</>
}

function applyEmphasis(
  s: string,
  baseKey: number,
  _k: number
): React.ReactNode[] {
  // Minimal: *bold* and _italic_, non-overlapping, no escaping. Good enough
  // for typical Slack messages; we accept that pathological inputs render
  // verbatim.
  const out: React.ReactNode[] = []
  const re = /(\*[^*]+\*|_[^_]+_)/g
  let last = 0
  let m: RegExpExecArray | null
  let i = 0
  while ((m = re.exec(s)) !== null) {
    if (m.index > last) out.push(s.slice(last, m.index))
    const tok = m[0]
    if (tok.startsWith('*')) {
      out.push(<strong key={`b-${baseKey}-${i++}`}>{tok.slice(1, -1)}</strong>)
    } else {
      out.push(<em key={`i-${baseKey}-${i++}`}>{tok.slice(1, -1)}</em>)
    }
    last = m.index + tok.length
  }
  if (last < s.length) out.push(s.slice(last))
  return out
}
```

```tsx
// app/workspace/[workspaceUrl]/components/TaskDetailTimeline/AddSlackContextDialog.tsx
'use client'

import { useState } from 'react'
import { useIntl } from 'react-intl'

interface Props {
  open: boolean
  onClose: () => void
  onSubmit: (permalink: string) => Promise<void>
}

const ERROR_CODE_TO_KEY: Record<string, string> = {
  invalid_permalink: 'slackContext.error.invalidPermalink',
  not_connected: 'slackContext.error.notConnected',
  wrong_team: 'slackContext.error.wrongTeam',
  not_in_channel: 'slackContext.error.notInChannel',
  message_gone: 'slackContext.error.messageGone',
  slack_unavailable: 'slackContext.error.slackUnavailable',
}

export function AddSlackContextDialog({
  open,
  onClose,
  onSubmit,
}: Props): React.ReactElement | null {
  const intl = useIntl()
  const [permalink, setPermalink] = useState('')
  const [errorMsg, setErrorMsg] = useState<string | null>(null)
  const [submitting, setSubmitting] = useState(false)

  if (!open) return null

  const handleSubmit = async () => {
    setSubmitting(true)
    setErrorMsg(null)
    try {
      await onSubmit(permalink)
      setPermalink('')
      onClose()
    } catch (e) {
      const code = e instanceof Error ? e.message : 'slack_unavailable'
      const key =
        ERROR_CODE_TO_KEY[code] ?? 'slackContext.error.slackUnavailable'
      setErrorMsg(intl.formatMessage({ id: key, defaultMessage: code }))
    } finally {
      setSubmitting(false)
    }
  }

  return (
    <div className='fixed inset-0 z-50 flex items-center justify-center bg-black/40'>
      <div className='bg-white dark:bg-zinc-900 rounded-md shadow-lg p-4 w-[480px] max-w-[90vw]'>
        <h2 className='text-sm font-semibold mb-3'>
          {intl.formatMessage({
            id: 'slackContext.dialog.title',
            defaultMessage: 'Attach a Slack message',
          })}
        </h2>
        <label className='block text-xs text-zinc-500 mb-1'>
          {intl.formatMessage({
            id: 'slackContext.dialog.permalinkLabel',
            defaultMessage: 'Slack permalink',
          })}
        </label>
        <input
          autoFocus
          value={permalink}
          onChange={e => setPermalink(e.target.value)}
          placeholder={intl.formatMessage({
            id: 'slackContext.dialog.permalinkPlaceholder',
            defaultMessage: '',
          })}
          className='w-full text-sm border border-zinc-300 dark:border-zinc-700 rounded px-2 py-1 bg-transparent'
        />
        {errorMsg && <p className='mt-2 text-xs text-red-500'>{errorMsg}</p>}
        <div className='mt-4 flex justify-end gap-2'>
          <button
            onClick={onClose}
            className='text-sm px-3 py-1 text-zinc-500 hover:text-zinc-700'
          >
            {intl.formatMessage({
              id: 'common.cancel',
              defaultMessage: 'Cancel',
            })}
          </button>
          <button
            onClick={handleSubmit}
            disabled={submitting || permalink.length === 0}
            className='text-sm px-3 py-1 rounded bg-zinc-800 text-white dark:bg-zinc-200 dark:text-zinc-900 disabled:opacity-50'
          >
            {intl.formatMessage({
              id: 'slackContext.dialog.submit',
              defaultMessage: 'Attach',
            })}
          </button>
        </div>
      </div>
    </div>
  )
}
```
