# Drafts ("草稿箱") Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a drafts system with DB-backed explicit drafts, localStorage form recovery, a dedicated drafts page, and sidebar navigation entry.

**Architecture:** New `Draft` Prisma model stores serialized form data as JSON. Three API routes handle CRUD. A new `/drafts` page aggregates Draft records and `isIdea` projects. All three creation modals (NewTaskModal, NewProjectModal, QuickDispatchModal) get "Save Draft" buttons and localStorage auto-recovery banners. The sidebar gets a "Drafts" entry with a count badge.

**Tech Stack:** Next.js 14 App Router, Prisma, Zod, TypeScript, react-intl, sonner (toast), localStorage, shadcn/ui, lucide-react

```typescript
import { NextRequest, NextResponse } from 'next/server'
import { requireAuth } from '@/lib/auth'
import { requireWorkspaceByUrl, zodError } from '@/lib/api/helpers'
import { prisma } from '@/lib/db/prisma'
import {
  upsertDraftSchema,
  taskDraftDataSchema,
  projectDraftDataSchema,
} from '@/lib/validation/schemas'

export async function GET(request: NextRequest) {
  try {
    const authResult = await requireAuth()
    if (authResult instanceof NextResponse) return authResult
    const { userId } = authResult

    const workspaceUrl = request.nextUrl.searchParams.get('workspaceUrl')
    const wsResult = await requireWorkspaceByUrl(userId, workspaceUrl)
    if (wsResult instanceof NextResponse) return wsResult

    const { member } = wsResult

    // Get the actor for this member
    const actor = await prisma.actor.findUnique({
      where: { memberId: member.id },
    })
    if (!actor) {
      return NextResponse.json({ drafts: [] })
    }

    const drafts = await prisma.draft.findMany({
      where: {
        workspaceId: wsResult.workspace.id,
        creatorId: actor.id,
      },
      orderBy: { updatedAt: 'desc' },
      select: {
        id: true,
        type: true,
        data: true,
        createdAt: true,
        updatedAt: true,
      },
    })

    return NextResponse.json({ drafts })
  } catch {
    return NextResponse.json(
      { error: 'Internal Server Error' },
      { status: 500 }
    )
  }
}

export async function POST(request: NextRequest) {
  try {
    const authResult = await requireAuth()
    if (authResult instanceof NextResponse) return authResult
    const { userId } = authResult

    const body = await request.json()
    const parsed = upsertDraftSchema.safeParse(body)
    if (!parsed.success) return zodError(parsed)

    const { id, workspaceUrl, type, data } = parsed.data

    // Validate data shape based on type
    const dataSchema =
      type === 'TASK' ? taskDraftDataSchema : projectDraftDataSchema
    const dataParsed = dataSchema.safeParse(data)
    if (!dataParsed.success) return zodError(dataParsed)

    const wsResult = await requireWorkspaceByUrl(userId, workspaceUrl)
    if (wsResult instanceof NextResponse) return wsResult

    const { member } = wsResult

    const actor = await prisma.actor.findUnique({
      where: { memberId: member.id },
    })
    if (!actor) {
      return NextResponse.json({ error: 'Actor not found' }, { status: 404 })
    }

    let draft
    if (id) {
      // Update existing draft — verify ownership
      const existing = await prisma.draft.findUnique({ where: { id } })
      if (!existing || existing.creatorId !== actor.id) {
        return NextResponse.json({ error: 'Draft not found' }, { status: 404 })
      }
      draft = await prisma.draft.update({
        where: { id },
        data: { data: dataParsed.data, type },
        select: {
          id: true,
          type: true,
          data: true,
          createdAt: true,
          updatedAt: true,
        },
      })
    } else {
      // Create new draft
      draft = await prisma.draft.create({
        data: {
          workspaceId: wsResult.workspace.id,
          creatorId: actor.id,
          type,
          data: dataParsed.data,
        },
        select: {
          id: true,
          type: true,
          data: true,
          createdAt: true,
          updatedAt: true,
        },
      })
    }

    return NextResponse.json({ draft }, { status: id ? 200 : 201 })
  } catch {
    return NextResponse.json(
      { error: 'Internal Server Error' },
      { status: 500 }
    )
  }
}
```

```typescript
import { NextRequest, NextResponse } from 'next/server'
import { requireAuth } from '@/lib/auth'
import { requireWorkspaceByUrl } from '@/lib/api/helpers'
import { prisma } from '@/lib/db/prisma'

type Params = { params: Promise<{ id: string }> }

export async function DELETE(request: NextRequest, { params }: Params) {
  try {
    const authResult = await requireAuth()
    if (authResult instanceof NextResponse) return authResult
    const { userId } = authResult

    const { id } = await params

    const workspaceUrl = request.nextUrl.searchParams.get('workspaceUrl')
    const wsResult = await requireWorkspaceByUrl(userId, workspaceUrl)
    if (wsResult instanceof NextResponse) return wsResult

    const { member } = wsResult

    const actor = await prisma.actor.findUnique({
      where: { memberId: member.id },
    })
    if (!actor) {
      return NextResponse.json({ error: 'Actor not found' }, { status: 404 })
    }

    const draft = await prisma.draft.findUnique({ where: { id } })
    if (!draft || draft.creatorId !== actor.id) {
      return NextResponse.json({ error: 'Draft not found' }, { status: 404 })
    }

    await prisma.draft.delete({ where: { id } })

    return NextResponse.json({ success: true })
  } catch {
    return NextResponse.json(
      { error: 'Internal Server Error' },
      { status: 500 }
    )
  }
}
```

```typescript
'use client'

import { useEffect, useRef, useCallback, useState } from 'react'
import { readCache, writeCache } from '../cache'

interface UseFormRecoveryOptions<T extends Record<string, unknown>> {
  /** localStorage cache key, e.g. CACHE_KEY.formRecoveryNewTask */
  cacheKey: string
  /** If true, skip auto-save (e.g., editing a DB draft). */
  disabled?: boolean | undefined
  /** Returns true if all fields are at default/empty values. */
  isEmpty: (data: T) => boolean
}

interface UseFormRecoveryReturn<T> {
  /** Cached form data from a previous session, or null. */
  recovered: T | null
  /** Call when user clicks "Restore". Clears recovered so banner hides. */
  acceptRecovery: () => T | null
  /** Call when user clicks "Discard". Clears both recovered state and localStorage. */
  discardRecovery: () => void
  /** Call on every form change (debounced internally). */
  save: (data: T) => void
  /** Call on submit or "Save Draft" to clear localStorage. */
  clear: () => void
}

export function useFormRecovery<T extends Record<string, unknown>>({
  cacheKey,
  disabled = false,
  isEmpty,
}: UseFormRecoveryOptions<T>): UseFormRecoveryReturn<T> {
  const [recovered, setRecovered] = useState<T | null>(() => {
    if (disabled) return null
    return readCache<T>(cacheKey)
  })

  const timerRef = useRef<ReturnType<typeof setTimeout> | null>(null)

  // Cleanup debounce timer on unmount
  useEffect(() => {
    return () => {
      if (timerRef.current) clearTimeout(timerRef.current)
    }
  }, [])

  const save = useCallback(
    (data: T) => {
      if (disabled) return
      if (timerRef.current) clearTimeout(timerRef.current)
      timerRef.current = setTimeout(() => {
        if (isEmpty(data)) {
          localStorage.removeItem(cacheKey)
        } else {
          writeCache(cacheKey, data)
        }
      }, 500)
    },
    [cacheKey, disabled, isEmpty]
  )

  const clear = useCallback(() => {
    if (timerRef.current) clearTimeout(timerRef.current)
    localStorage.removeItem(cacheKey)
    setRecovered(null)
  }, [cacheKey])

  const acceptRecovery = useCallback(() => {
    const data = recovered
    setRecovered(null)
    // Don't clear localStorage yet — user might close without submitting
    return data
  }, [recovered])

  const discardRecovery = useCallback(() => {
    localStorage.removeItem(cacheKey)
    setRecovered(null)
  }, [cacheKey])

  return { recovered, acceptRecovery, discardRecovery, save, clear }
}
```

```typescript
'use client'

import { useState, useCallback } from 'react'
import { useParams } from 'next/navigation'
import { apiFetch } from '../utils'
import { API } from '../constants'
import type { ApiDraft } from '../types'

interface UseDraftsReturn {
  drafts: ApiDraft[]
  loading: boolean
  error: string | null
  draftCount: number
  fetchDrafts: () => Promise<void>
  saveDraft: (
    type: 'TASK' | 'PROJECT',
    data: Record<string, unknown>,
    id?: string | undefined
  ) => Promise<ApiDraft | null>
  deleteDraft: (id: string) => Promise<boolean>
}

export function useDrafts(): UseDraftsReturn {
  const { workspaceUrl } = useParams<{ workspaceUrl: string }>()
  const [drafts, setDrafts] = useState<ApiDraft[]>([])
  const [loading, setLoading] = useState(false)
  const [error, setError] = useState<string | null>(null)

  const fetchDrafts = useCallback(async () => {
    setLoading(true)
    setError(null)
    try {
      const res = await apiFetch(
        `${API.drafts.list}?workspaceUrl=${encodeURIComponent(workspaceUrl)}`
      )
      if (res.ok) {
        const data = await res.json()
        setDrafts(data.drafts ?? [])
      } else {
        setError('Failed to load drafts')
      }
    } catch {
      setError('Failed to load drafts')
    } finally {
      setLoading(false)
    }
  }, [workspaceUrl])

  const saveDraft = useCallback(
    async (
      type: 'TASK' | 'PROJECT',
      data: Record<string, unknown>,
      id?: string | undefined
    ): Promise<ApiDraft | null> => {
      try {
        const res = await apiFetch(API.drafts.list, {
          method: 'POST',
          body: { id, workspaceUrl, type, data },
        })
        if (res.ok) {
          const result = await res.json()
          const draft: ApiDraft = result.draft
          // Update local list
          setDrafts(prev => {
            const filtered = prev.filter(d => d.id !== draft.id)
            return [draft, ...filtered]
          })
          return draft
        }
        return null
      } catch {
        return null
      }
    },
    [workspaceUrl]
  )

  const deleteDraft = useCallback(
    async (id: string): Promise<boolean> => {
      try {
        const res = await apiFetch(
          `${API.drafts.byId(id)}?workspaceUrl=${encodeURIComponent(workspaceUrl)}`,
          { method: 'DELETE' }
        )
        if (res.ok) {
          setDrafts(prev => prev.filter(d => d.id !== id))
          return true
        }
        return false
      } catch {
        return false
      }
    },
    [workspaceUrl]
  )

  return {
    drafts,
    loading,
    error,
    draftCount: drafts.length,
    fetchDrafts,
    saveDraft,
    deleteDraft,
  }
}
```
