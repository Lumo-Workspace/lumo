# Sprints Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Spec:** [`docs/superpowers/specs/2026-04-19-sprints-design.md`](../specs/2026-04-19-sprints-design.md)

**Goal:** Add Team-level sprints to Lumo — fixed-cadence iterations with auto-prepared drafts, forced-choice spillover on close, and an LLM-generated retro report. No story points/velocity/burndown in v1.

**Architecture:** New `Sprint` and `SprintSummary` Prisma models. `Task.sprintId` nullable; null = backlog. Single active sprint per team enforced by a partial unique index plus service-layer checks inside a `SELECT … FOR UPDATE` sprint-row lock. Service layer (`sprint.service`, `sprint-prepare.service`, `sprint-summary.service`) sits between thin API routes and repositories. UI is a new top-level "Sprints" sidebar entry with a landing page (per-team active/next/recent) and a sprint detail page that reuses the existing Kanban component.

**Tech Stack:** Prisma (PostgreSQL), Next.js 14 App Router, TypeScript strict, Clerk auth, Zod validation, Jest, Tailwind + shadcn/ui, Vercel Cron.

**Phasing:** Four PRs. Each is independently deployable; each has its own commit + PR at the end of its phase section.

// lib/repositories/sprint.repository.ts
import { prisma } from '@/lib/db/prisma'
import type { SprintStatus, Prisma } from '@prisma/client'

export interface CreateSprintInput {
  teamId: string
  number: number
  name: string
  startDate: Date
  endDate: Date
}

export interface UpdateSprintInput {
  name?: string
  startDate?: Date
  endDate?: Date
  status?: SprintStatus
  startedAt?: Date | null
  closedAt?: Date | null
}

export const sprintRepository = {
  async findById(sprintId: string) {
    return prisma.sprint.findUnique({
      where: { id: sprintId },
      include: {
        team: {
          select: {
            id: true,
            name: true,
            workspaceId: true,
            sprintCadenceDays: true,
          },
        },
      },
    })
  },

  async findActiveByTeam(teamId: string) {
    return prisma.sprint.findFirst({
      where: { teamId, status: 'ACTIVE' },
    })
  },

  async listByTeam(
    teamId: string,
    opts: { status?: SprintStatus; limit?: number; cursor?: string } = {}
  ) {
    const { status, limit = 20, cursor } = opts
    return prisma.sprint.findMany({
      where: { teamId, ...(status ? { status } : {}) },
      orderBy: { number: 'desc' },
      take: limit,
      ...(cursor ? { cursor: { id: cursor }, skip: 1 } : {}),
    })
  },

  async nextNumberForTeam(teamId: string): Promise<number> {
    const agg = await prisma.sprint.aggregate({
      where: { teamId },
      _max: { number: true },
    })
    return (agg._max.number ?? 0) + 1
  },

  async create(input: CreateSprintInput) {
    return prisma.sprint.create({ data: input })
  },

  async update(sprintId: string, data: UpdateSprintInput) {
    return prisma.sprint.update({ where: { id: sprintId }, data })
  },

  /** Row lock inside a transaction. Used for serialising start/close/task-membership. */
  async lockForUpdate(tx: Prisma.TransactionClient, sprintId: string) {
    const rows = await tx.$queryRaw<{ id: string; status: SprintStatus }[]>`
      SELECT id, status FROM sprints WHERE id = ${sprintId} FOR UPDATE
    `
    return rows[0] ?? null
  },

  async remove(sprintId: string) {
    return prisma.sprint.delete({ where: { id: sprintId } })
  },

  async countTasks(sprintId: string, filter: { status?: string } = {}) {
    return prisma.task.count({
      where: {
        sprintId,
        ...(filter.status ? { status: filter.status as any } : {}),
      },
    })
  },
}
```

import { prisma } from '@/lib/db/prisma'
import type { Prisma } from '@prisma/client'

export interface SprintSummaryData {
  stats: Prisma.InputJsonValue
  tldr: string
  report: string
}

export const sprintSummaryRepository = {

    return prisma.sprintSummary.findUnique({ where: { sprintId } })

    return prisma.sprintSummary.upsert({
      where: { sprintId },
      create: { sprintId, ...data },
      update: { ...data, generatedAt: new Date() },
    })
  },
}

  throw Object.assign(new Error(message), { status })

export interface CreateDraftInput {
  name?: string
  startDate?: Date
  endDate?: Date
}

export const sprintService = {
  async createDraft(
    teamId: string,
    input: CreateDraftInput,
    _actor: Pick<Actor, 'memberId'>
  ) {
    const team = await prisma.team.findUnique({
      where: { id: teamId },
      select: { id: true, sprintCadenceDays: true },
    })
    if (!team) httpError(404, 'team not found')

    let startDate = input.startDate
    let endDate = input.endDate

    if (!startDate || !endDate) {
      if (!team.sprintCadenceDays) {
        httpError(
          400,
          'team has no cadence; startDate and endDate are required'
        )
      }
      if (!startDate) startDate = nextUtcMondayMidnight(new Date())
      if (!endDate) endDate = addDays(startDate, team.sprintCadenceDays)
    }

    const number = await sprintRepository.nextNumberForTeam(teamId)
    const name = input.name?.trim() || `Sprint ${number}`

    return sprintRepository.create({ teamId, number, name, startDate, endDate })
  },
}

export function nextUtcMondayMidnight(from: Date): Date {
  const d = new Date(
    Date.UTC(from.getUTCFullYear(), from.getUTCMonth(), from.getUTCDate())
  )
  const day = d.getUTCDay() // 0 Sun … 6 Sat
  const delta = (8 - day) % 7 || 7
  d.setUTCDate(d.getUTCDate() + delta)
  return d
}

export function addDays(d: Date, days: number): Date {
  const r = new Date(d)
  r.setUTCDate(r.getUTCDate() + days)
  return r
}

type SpilloverDecision = 'moveToNext' | 'backToBacklog'

export type SprintCloseDecisions = Record<string, SpilloverDecision>

async close(
  sprintId: string,
  decisions: SprintCloseDecisions,
  actor: Pick<Actor, 'memberId'> & { id?: string }
) {
  const sprint = await sprintRepository.findById(sprintId)
  if (!sprint) httpError(404, 'sprint not found')

  return prisma.$transaction(async tx => {
    const locked = await sprintRepository.lockForUpdate(tx, sprintId)
    if (!locked) httpError(404, 'sprint not found')
    if (locked.status !== 'ACTIVE') httpError(409, `cannot close sprint in ${locked.status} state`)

    const unfinished = await (tx as any).task.findMany({
      where: { sprintId, status: { not: 'DONE' } },
      select: { id: true, status: true },
    })
    const unfinishedIds = new Set(unfinished.map((t: any) => t.id))
    const decisionIds = new Set(Object.keys(decisions))

    // Every unfinished task must have a decision
    for (const id of unfinishedIds) {
      if (!decisionIds.has(id as string)) {
        httpError(400, `decision missing for task ${id}`)
      }
    }
    // No stale decisions (task no longer in sprint)
    for (const id of decisionIds) {
      if (!unfinishedIds.has(id as string)) {
        httpError(409, 'sprint membership changed; reopen close dialog')
      }
    }

    // Ensure / create next DRAFT if any decision is moveToNext
    let nextSprintId: string | null = null
    const needsNext = Object.values(decisions).some(d => d === 'moveToNext')
    if (needsNext) {
      // Try existing DRAFT after current
      const drafts = await sprintRepository.listByTeam(sprint.teamId, { status: 'DRAFT', limit: 1 })
      if (drafts.length > 0) {
        nextSprintId = drafts[0].id
      } else {
        const nextNumber = await sprintRepository.nextNumberForTeam(sprint.teamId)
        const teamRow = await (tx as any).team.findUnique({
          where: { id: sprint.teamId },
          select: { sprintCadenceDays: true },
        })
        const cadence = teamRow?.sprintCadenceDays ?? 14
        const startDate = sprint.endDate > new Date() ? sprint.endDate : new Date()
        const endDate = addDays(startDate, cadence)
        const created = await sprintRepository.create({
          teamId: sprint.teamId,
          number: nextNumber,
          name: `Sprint ${nextNumber}`,
          startDate,
          endDate,
        })
        nextSprintId = created.id
      }
    }

    // Apply decisions in two bulk updates
    const toNext: string[] = []
    const toBacklog: string[] = []
    for (const [taskId, d] of Object.entries(decisions)) {
      if (d === 'moveToNext') toNext.push(taskId)
      else toBacklog.push(taskId)
    }
    if (toNext.length) {
      await taskRepository.bulkUpdateSprint(toNext, nextSprintId!)
    }
    if (toBacklog.length) {
      await taskRepository.bulkUpdateSprint(toBacklog, null)
    }

    // Activity rows — one per moved task
    const actorId = actor.id ?? null
    await (tx as any).taskActivity.createMany({
      data: [
        ...toNext.map(taskId => ({
          taskId,
          actorId,
          type: 'SPRINT_CHANGED' as const,
          payload: { fromSprintId: sprintId, toSprintId: nextSprintId, reason: 'spillover' },
        })),
        ...toBacklog.map(taskId => ({
          taskId,
          actorId,
          type: 'SPRINT_CHANGED' as const,
          payload: { fromSprintId: sprintId, toSprintId: null, reason: 'spillover' },
        })),
      ],
      skipDuplicates: true,
    })

    const closed = await sprintRepository.update(sprintId, {
      status: 'CLOSED',
      closedAt: new Date(),
    })

    // Enqueue retro summary generation
    await (tx as any).generationJob.create({
      data: { type: 'sprint', targetId: sprintId, status: 'pending' },
    })

    return closed
  })
},

async addTaskToSprint(taskId: string, sprintId: string, actor: Pick<Actor, 'memberId'> & { id?: string }) {
  const [task, sprint] = await Promise.all([
    prisma.task.findUnique({ where: { id: taskId }, select: { id: true, teamId: true, sprintId: true } }),
    sprintRepository.findById(sprintId),
  ])
  if (!task) httpError(404, 'task not found')
  if (!sprint) httpError(404, 'sprint not found')
  if (task.teamId !== sprint.teamId) httpError(400, 'task and sprint belong to different teams')
  if (sprint.status === 'CLOSED') httpError(409, 'sprint is closed')

  return prisma.$transaction(async tx => {
    await sprintRepository.lockForUpdate(tx, sprintId)
    const fromSprintId = task.sprintId
    await taskRepository.bulkUpdateSprint([taskId], sprintId)
    await (tx as any).taskActivity.create({
      data: {
        taskId,
        actorId: actor.id ?? null,
        type: 'SPRINT_CHANGED',
        payload: { fromSprintId, toSprintId: sprintId, reason: 'planning' },
      },
    })
    return { taskId, sprintId }
  })
},

async removeTaskFromSprint(taskId: string, actor: Pick<Actor, 'memberId'> & { id?: string }) {
  const task = await prisma.task.findUnique({
    where: { id: taskId },
    select: { id: true, sprintId: true },

  if (!task) httpError(404, 'task not found')
  if (!task.sprintId) return { taskId, sprintId: null }

  const sprint = await sprintRepository.findById(task.sprintId)
  if (sprint && sprint.status === 'CLOSED') {

  return prisma.$transaction(async tx => {

      data: {
        taskId,
        actorId: actor.id ?? null,
        type: 'SPRINT_CHANGED',
        payload: { fromSprintId: task.sprintId, toSprintId: null, reason: 'planning' },
      },
    })
    return { taskId, sprintId: null }
  })
},

// lib/services/sprint-prepare.service.ts
import { prisma } from '@/lib/db/prisma'
import { sprintRepository } from '@/lib/repositories/sprint.repository'
import { sprintService } from '@/lib/services/sprint.service'
import { addDays, nextUtcMondayMidnight } from '@/lib/services/sprint.service'

const SYSTEM_ACTOR = { memberId: 'system' } as const

export const sprintPrepareService = {
  async prepareNextDraft(teamId: string): Promise<void> {
    const team = await prisma.team.findUnique({
      where: { id: teamId },
      select: { id: true, sprintCadenceDays: true, sprintAutoPrepare: true },
    })
    if (!team || !team.sprintCadenceDays || !team.sprintAutoPrepare) return

    const existingDrafts = await sprintRepository.listByTeam(teamId, {
      status: 'DRAFT',
      limit: 1,
    })
    if (existingDrafts.length > 0) return

    const latestAny = await sprintRepository.listByTeam(teamId, { limit: 1 })
    const startDate =
      latestAny.length > 0
        ? latestAny[0].endDate
        : nextUtcMondayMidnight(new Date())
    const endDate = addDays(startDate, team.sprintCadenceDays)

    await sprintService.createDraft(
      teamId,
      { startDate, endDate },
      SYSTEM_ACTOR as any
    )
  },

  async prepareAll(): Promise<{ processed: number }> {
    const teams = await prisma.team.findMany({
      where: { sprintCadenceDays: { not: null }, sprintAutoPrepare: true },
      select: { id: true, sprintCadenceDays: true, sprintAutoPrepare: true },
    })
    for (const team of teams) {
      try {
        await sprintPrepareService.prepareNextDraft(team.id)
      } catch (e) {
        // swallow per-team errors to let the rest proceed; log for observability
        console.error(`[sprint-prepare] team ${team.id} failed`, e)
      }
    }
    return { processed: teams.length }
  },
}

import { sprintRepository } from '@/lib/repositories/sprint.repository'
import { sprintSummaryRepository } from '@/lib/repositories/sprint-summary.repository'
// If the workspace already exposes a shared helper, prefer it:
import { callSummaryLLM } from '@/lib/llm/summary-client'

function httpError(status: number, message: string): never {
  throw Object.assign(new Error(message), { status })
}

export interface SprintStats {
  totalTasks: number
  done: number
  spillover: number
  byPriority: Record<string, number>
  byAssignee: Record<string, number>
}

export const sprintSummaryService = {
  async generate(sprintId: string) {
    const sprint = await sprintRepository.findById(sprintId)
    if (!sprint) httpError(404, 'sprint not found')
    if (sprint.status !== 'CLOSED') httpError(400, 'sprint is not closed')

    const tasks = await prisma.task.findMany({
      where: {
        OR: [
          { sprintId }, // still in sprint (done + kept)
          // tasks moved out via spillover are no longer `sprintId = this one` — infer from TaskActivity
        ],
      },
      select: {
        id: true,
        status: true,
        priority: true,
        assigneeActorId: true,
        title: true,
      },
    })

    const spilloverActivities = await prisma.taskActivity.findMany({
      where: {
        type: 'SPRINT_CHANGED',
        payload: { path: ['fromSprintId'], equals: sprintId },
      },
      select: { taskId: true, payload: true },
    })
    const spilloverCount = spilloverActivities.length

    const stats: SprintStats = {
      totalTasks: tasks.length + spilloverCount,
      done: tasks.filter(t => t.status === 'DONE').length,
      spillover: spilloverCount,
      byPriority: tasks.reduce<Record<string, number>>((acc, t) => {
        acc[t.priority] = (acc[t.priority] ?? 0) + 1
        return acc
      }, {}),
      byAssignee: tasks.reduce<Record<string, number>>((acc, t) => {
        const key = t.assigneeActorId ?? 'unassigned'
        acc[key] = (acc[key] ?? 0) + 1
        return acc
      }, {}),
    }

    const { tldr, report } = await callSummaryLLM({
      kind: 'sprint_retro',
      subject: sprint.name,
      window: { start: sprint.startDate, end: sprint.endDate },
      stats,
      tasks,
    })

    return sprintSummaryRepository.save(sprintId, {
      stats: stats as any,
      tldr,
      report,
    })
  },

// lib/auth/require-team-access.ts
import { prisma } from '@/lib/db/prisma'

export async function requireTeamMemberOrOwner(
  workspaceId: string,
  teamId: string,
  clerkId: string
): Promise<{ memberId: string; reason: 'owner' | 'team' }> {
  const member = await prisma.workspaceMember.findUnique({
    where: { workspaceId_clerkId: { workspaceId, clerkId } },
    select: { id: true, role: true },
  })
  if (!member) {
    throw Object.assign(new Error('not a workspace member'), { status: 403 })
  }
  if (member.role === 'OWNER') {
    return { memberId: member.id, reason: 'owner' }
  }
  const tm = await prisma.teamMember.findUnique({
    where: { teamId_memberId: { teamId, memberId: member.id } },
    select: { id: true },
  })
  if (!tm) {
    throw Object.assign(new Error('not a team member'), { status: 403 })
  }
  return { memberId: member.id, reason: 'team' }
}

// lib/api-client/sprints.ts
import type { Sprint, SprintStatus, SprintSummary } from '@prisma/client'
import type { SprintCloseDecisions } from '@/lib/validation/sprint.schema'

interface ApiError extends Error {
  status: number
  body: unknown
}

async function request<T>(input: RequestInfo, init?: RequestInit): Promise<T> {
  const res = await fetch(input, {
    ...init,
    headers: { 'Content-Type': 'application/json', ...(init?.headers ?? {}) },
  })
  if (!res.ok) {
    const body = await res.json().catch(() => ({}))
    const err = new Error(body?.error ?? res.statusText) as ApiError
    err.status = res.status
    err.body = body
    throw err
  }
  if (res.status === 204) return undefined as T
  return res.json() as Promise<T>
}

export const sprintsApi = {
  listByTeam(
    teamId: string,
    opts: { status?: SprintStatus; cursor?: string; limit?: number } = {}
  ) {
    const q = new URLSearchParams()
    if (opts.status) q.set('status', opts.status)
    if (opts.cursor) q.set('cursor', opts.cursor)
    if (opts.limit) q.set('limit', String(opts.limit))
    return request<{ sprints: Sprint[] }>(
      `/api/teams/${teamId}/sprints?${q.toString()}`
    )
  },

  createDraft(
    teamId: string,
    input: { name?: string; startDate?: string; endDate?: string }
  ) {
    return request<Sprint>(`/api/teams/${teamId}/sprints`, {
      method: 'POST',
      body: JSON.stringify(input),
    })
  },

  get(sprintId: string) {
    return request<{
      sprint: Sprint
      progress: { done: number; total: number }
    }>(`/api/sprints/${sprintId}`)
  },

  update(
    sprintId: string,
    input: { name?: string; startDate?: string; endDate?: string }
  ) {
    return request<Sprint>(`/api/sprints/${sprintId}`, {
      method: 'PATCH',
      body: JSON.stringify(input),
    })
  },

  remove(sprintId: string) {
    return request<void>(`/api/sprints/${sprintId}`, { method: 'DELETE' })
  },

  start(sprintId: string) {
    return request<Sprint>(`/api/sprints/${sprintId}/start`, { method: 'POST' })
  },

  close(sprintId: string, decisions: SprintCloseDecisions) {
    return request<Sprint>(`/api/sprints/${sprintId}/close`, {
      method: 'POST',
      body: JSON.stringify({ decisions }),
    })

// app/workspace/[workspaceUrl]/sprints/components/CloseSprintDialog.tsx
'use client'
import { useEffect, useState } from 'react'
import {
  Dialog,
  DialogContent,
  DialogHeader,
  DialogTitle,
  DialogFooter,
} from '@/components/ui/dialog'
import { Button } from '@/components/ui/button'
import { RadioGroup, RadioGroupItem } from '@/components/ui/radio-group'
import { Label } from '@/components/ui/label'
import { sprintsApi } from '@/lib/api-client/sprints'
import { mutate } from 'swr'

type Decision = 'moveToNext' | 'backToBacklog'

export function CloseSprintDialog({
  sprintId,
  onClose,
}: {
  sprintId: string
  onClose: () => void
}) {
  const [tasks, setTasks] = useState<
    Array<{ id: string; title: string; status: string }>
  >([])
  const [decisions, setDecisions] = useState<Record<string, Decision>>({})
  const [submitting, setSubmitting] = useState(false)

  useEffect(() => {
    // Fetch the currently unfinished tasks in this sprint via existing task-list API,
    // e.g. fetch(`/api/tasks?sprintId=${sprintId}&statusNot=DONE`).
    // Adjust the call to whatever the project's task-list endpoint accepts.
    fetch(`/api/tasks?sprintId=${sprintId}&statusNot=DONE`)
      .then(r => r.json())
      .then(body => setTasks(body.tasks ?? []))
  }, [sprintId])

  const allDecided = tasks.length > 0 && tasks.every(t => decisions[t.id])

  function setDecisionFor(taskId: string, d: Decision) {
    setDecisions(prev => ({ ...prev, [taskId]: d }))
  }
  function bulk(d: Decision) {
    const next: Record<string, Decision> = {}
    for (const t of tasks) next[t.id] = d
    setDecisions(next)
  }

  async function submit() {
    setSubmitting(true)
    try {
      await sprintsApi.close(sprintId, decisions)
      mutate(['sprint', sprintId])
      onClose()
    } catch (err: any) {
      if (err.status === 409) {
        alert('Sprint membership changed; reloading task list.')
        const res = await fetch(
          `/api/tasks?sprintId=${sprintId}&statusNot=DONE`
        )
        setTasks((await res.json()).tasks ?? [])
        setDecisions({})
      } else {
        alert(err.message)
      }
    } finally {
      setSubmitting(false)
    }
  }

  return (
    <Dialog open onOpenChange={onClose}>
      <DialogContent className='max-w-lg'>
        <DialogHeader>
          <DialogTitle>Close sprint?</DialogTitle>
        </DialogHeader>
        {tasks.length === 0 ? (
          <p>All tasks complete — safe to close.</p>
        ) : (
          <>
            <p className='text-sm mb-2'>
              {tasks.length} unfinished task(s). Decide for each:
            </p>
            <div className='flex gap-2 mb-3'>
              <Button
                size='sm'
                variant='outline'
                onClick={() => bulk('moveToNext')}
              >
                Move all to next
              </Button>
              <Button
                size='sm'
                variant='outline'
                onClick={() => bulk('backToBacklog')}
              >
                All to backlog
              </Button>
            </div>
            <div className='divide-y border rounded max-h-80 overflow-y-auto'>
              {tasks.map(t => (
                <div key={t.id} className='p-3'>
                  <p className='font-medium text-sm mb-1'>{t.title}</p>
                  <RadioGroup
                    value={decisions[t.id] ?? ''}
                    onValueChange={v => setDecisionFor(t.id, v as Decision)}
                    className='flex gap-4'
                  >
                    <div className='flex items-center gap-1'>
                      <RadioGroupItem value='moveToNext' id={`${t.id}-next`} />
                      <Label htmlFor={`${t.id}-next`} className='text-xs'>
                        Move to next
                      </Label>
                    </div>
                    <div className='flex items-center gap-1'>
                      <RadioGroupItem
                        value='backToBacklog'
                        id={`${t.id}-back`}
                      />
                      <Label htmlFor={`${t.id}-back`} className='text-xs'>
                        Back to backlog
                      </Label>
                    </div>
                  </RadioGroup>
                </div>
              ))}
            </div>
            <p className='text-xs text-muted-foreground mt-3'>
              This is irreversible.
            </p>
          </>
        )}
        <DialogFooter>
          <Button variant='ghost' onClick={onClose}>
            Cancel
          </Button>
          <Button
            onClick={submit}
            disabled={(!allDecided && tasks.length > 0) || submitting}
          >
            Close sprint
          </Button>
        </DialogFooter>
      </DialogContent>
    </Dialog>
  )
}
```

    <article className='prose max-w-none'>
      <h2>Retro: {data.tldr}</h2>
