# Memory Management UI Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Give humans a UI to view, create, edit, delete, and promote (TASK→PROJECT) the Memory rows Claude accumulates, tied to task and project detail pages.

**Architecture:** Add two nullable columns to `Memory` (`updatedAt`, `createdByMemberId`); extend `memoryRepository` with list/update/find-by-id methods; a new `memoryManagementService` (disjoint from the layer1/2 LLM services) drives 4 new REST routes (`POST /api/projects/[id]/memories`, `POST /api/tasks/[id]/memories`, `PATCH /api/memories/[id]`, `DELETE /api/memories/[id]`); a `memoriesClient` + `useMemories` react-query hook feeds a `MemoryTab` (with a shadcn-Dialog create/edit form) embedded in both Task (`TimelineTabs`) and Project (`ProjectDetailPage`) detail pages. PATCH carries promote via `{ scope: 'PROJECT' }` — no separate verb. Delete is hard-delete.

**Tech Stack:** Next.js 14 App Router, TypeScript (strict), Prisma + Neon Postgres, Clerk + bearer (`requireIdentity`), Zod validation, @tanstack/react-query, shadcn/ui, react-intl (17 locales), Jest.

```ts
  /** All PROJECT-scope memories for a project, newest first. Management UI. */
  async listForProject(projectId: string) {
    return prisma.memory.findMany({
      where: { projectId, scope: 'PROJECT' },
      orderBy: { createdAt: 'desc' },
      select: MEMORY_ROW_SELECT,
    })
  },

  /** All memories for a task, newest first. Management UI. */
  async listForTask(taskId: string) {
    return prisma.memory.findMany({
      where: { taskId },
      orderBy: { createdAt: 'desc' },
      select: MEMORY_ROW_SELECT,
    })
  },

  /**
   * Patch content and/or scope and/or taskId in one write. Undefined fields
   * are omitted so a content-only edit never touches scope. Returns the full
   * row for the HTTP response.
   */
  async update(
    id: string,
    patch: {
      content?: Prisma.InputJsonValue
      scope?: MemoryScope
      taskId?: string | null
    },
    tx?: Prisma.TransactionClient
  ) {
    const client = tx ?? prisma
    const data: Prisma.MemoryUpdateInput = {}
    if (patch.content !== undefined) data.content = patch.content
    if (patch.scope !== undefined) data.scope = patch.scope
    if (patch.taskId !== undefined) {
      data.task =
        patch.taskId === null
          ? { disconnect: true }
          : { connect: { id: patch.taskId } }
    }
    return client.memory.update({
      where: { id },
      data,
      select: MEMORY_ROW_SELECT,
    })
  },

  /** Auth-relevant fields for PATCH/DELETE routes. */
  async findByIdWithWorkspace(id: string) {
    return prisma.memory.findUnique({
      where: { id },
      select: {
        id: true,
        workspaceId: true,
        projectId: true,
        taskId: true,
        scope: true,
        category: true,
      },
    })
  },
```

```ts
import { z } from 'zod'
import { MemoryCategory } from '@prisma/client'

/**
 * Body for POST /api/{projects,tasks}/[id]/memories. `content` is left as
 * unknown here because its required shape depends on `category`; the route
 * validates it a second time with parseMemoryContent(category, content).
 */
export const createMemoryBodySchema = z.object({
  category: z.nativeEnum(MemoryCategory),
  content: z.unknown().refine(v => v !== undefined, { message: 'Required' }),
})

/**
 * Body for PATCH /api/memories/[id]. At least one of content / scope. `scope`
 * may only be 'PROJECT' (promote); demote is out of scope.
 */
export const patchMemoryBodySchema = z
  .object({
    content: z.unknown().optional(),
    scope: z.literal('PROJECT').optional(),
  })
  .refine(b => b.content !== undefined || b.scope !== undefined, {
    message: 'At least one of content or scope must be provided',
  })

export type CreateMemoryBody = z.infer<typeof createMemoryBodySchema>
export type PatchMemoryBody = z.infer<typeof patchMemoryBodySchema>
```

```ts
import { memoryRepository } from '@/lib/repositories/memory.repository'
import type { MemoryCategory } from '@prisma/client'
import type { MemoryContent } from '@/lib/memory/schemas'
import type { Prisma } from '@prisma/client'

function httpError(message: string, status: number): Error {
  return Object.assign(new Error(message), { status })
}

interface CreateForProjectInput {
  projectId: string
  workspaceId: string
  category: MemoryCategory
  content: MemoryContent
  actorMemberId: string
}

interface CreateForTaskInput {
  taskId: string
  projectId: string
  workspaceId: string
  category: MemoryCategory
  content: MemoryContent
  actorMemberId: string
}

/**
 * Human-driven CRUD over Memory rows. Deliberately disjoint from
 * lib/services/memory/ (layer1/layer2), which are LLM-driven and must not
 * pull their gateway dependencies into the HTTP request path.
 */
export const memoryManagementService = {
  listProjectMemories(projectId: string) {
    return memoryRepository.listForProject(projectId)
  },

  listTaskMemories(taskId: string) {
    return memoryRepository.listForTask(taskId)
  },

  createForProject(input: CreateForProjectInput) {
    return memoryRepository.create({
      workspaceId: input.workspaceId,
      projectId: input.projectId,
      taskId: null,
      scope: 'PROJECT',
      category: input.category,
      content: input.content as Prisma.InputJsonValue,
      sourceSessionId: null,
      createdByMemberId: input.actorMemberId,
    })
  },

  createForTask(input: CreateForTaskInput) {
    return memoryRepository.create({
      workspaceId: input.workspaceId,
      projectId: input.projectId,
      taskId: input.taskId,
      scope: 'TASK',
      category: input.category,
      content: input.content as Prisma.InputJsonValue,
      sourceSessionId: null,
      createdByMemberId: input.actorMemberId,
    })
  },

  editContent(memoryId: string, content: MemoryContent) {
    return memoryRepository.update(memoryId, {
      content: content as Prisma.InputJsonValue,
    })
  },

  async promoteToProject(memoryId: string, content?: MemoryContent) {
    const existing = await memoryRepository.findByIdWithWorkspace(memoryId)
    if (!existing) throw httpError('Not found', 404)
    if (existing.scope === 'PROJECT') {
      throw httpError('Memory is already project-scoped', 409)
    }
    return memoryRepository.update(memoryId, {
      scope: 'PROJECT',
      taskId: null,
      ...(content !== undefined
        ? { content: content as Prisma.InputJsonValue }
        : {}),
    })
  },

  async delete(memoryId: string) {
    await memoryRepository.delete(memoryId)
  },
}
```

```ts
import { NextRequest, NextResponse } from 'next/server'
import { requireIdentity } from '@/lib/auth'
import { projectRepository } from '@/lib/repositories'
import { memoryManagementService } from '@/lib/services'
import { createMemoryBodySchema } from '@/lib/validation/memory'
import { parseMemoryContent } from '@/lib/memory/schemas'
import { zodError } from '@/lib/api/helpers'

export const runtime = 'nodejs'

type Params = { params: Promise<{ id: string }> }

export async function POST(request: NextRequest, { params }: Params) {
  try {
    const identity = await requireIdentity(request)
    if (identity instanceof NextResponse) return identity
    const { id } = await params

    const project = await projectRepository.findWorkspaceId(id)
    if (!project) return NextResponse.json({ error: 'Not found' }, { status: 404 })
    if (project.workspaceId !== identity.workspaceId) {
      return NextResponse.json({ error: 'Forbidden' }, { status: 403 })
    }

    const raw = await request.json()
    const parsed = createMemoryBodySchema.safeParse(raw)
    if (!parsed.success) return zodError(parsed)

    let content
    try {
      content = parseMemoryContent(parsed.data.category, parsed.data.content)
    } catch {
      return NextResponse.json(
        { error: 'Content does not match the category schema' },
        { status: 400 }
      )
    }

    const memory = await memoryManagementService.createForProject({
      projectId: id,
      workspaceId: identity.workspaceId,
      category: parsed.data.category,
      content,
      actorMemberId: identity.memberId,
    })
    return NextResponse.json({ memory }, { status: 201 })
  } catch (error) {
    if (error instanceof Error && 'status' in error) {
      return NextResponse.json(
        { error: error.message },
        { status: (error as Error & { status: number }).status }
      )
    }
    console.error('POST /api/projects/[id]/memories error:', error)
    return NextResponse.json({ error: 'Internal Server Error' }, { status: 500 })
  }
}
```

```ts
    }
```

```ts
    const parsed = patchMemoryBodySchema.safeParse(raw)
    if (!parsed.success) return zodError(parsed)

    let content: MemoryContent | undefined
    if (parsed.data.content !== undefined) {
      try {
        content = parseMemoryContent(existing.category, parsed.data.content)
      } catch {
        return NextResponse.json(
          { error: 'Content does not match the category schema' },
          { status: 400 }
        )
      }
    }

    let memory
    if (parsed.data.scope === 'PROJECT') {
      memory = await memoryManagementService.promoteToProject(memoryId, content)
    } else {
      // content is defined here (refine guarantees content-or-scope; scope absent ⇒ content present)
      memory = await memoryManagementService.editContent(
        memoryId,
        content as MemoryContent
      )
    }
    return NextResponse.json({ memory }, { status: 200 })
  } catch (error) {
    if (error instanceof Error && 'status' in error) {
      return NextResponse.json(
        { error: error.message },
        { status: (error as Error & { status: number }).status }
      )
    }
    console.error('PATCH /api/memories/[memoryId] error:', error)
    return NextResponse.json({ error: 'Internal Server Error' }, { status: 500 })
  }
}

export async function DELETE(request: NextRequest, { params }: Params) {
  try {
    const identity = await requireIdentity(request)
    if (identity instanceof NextResponse) return identity
    const { memoryId } = await params

    const existing = await memoryRepository.findByIdWithWorkspace(memoryId)
    if (!existing) return NextResponse.json({ error: 'Not found' }, { status: 404 })
    if (existing.workspaceId !== identity.workspaceId) {
      return NextResponse.json({ error: 'Forbidden' }, { status: 403 })
    }

    await memoryManagementService.delete(memoryId)
    return new NextResponse(null, { status: 204 })
  } catch (error) {
    if (error instanceof Error && 'status' in error) {
      return NextResponse.json(
        { error: error.message },
        { status: (error as Error & { status: number }).status }
      )
    }
    console.error('DELETE /api/memories/[memoryId] error:', error)
    return NextResponse.json({ error: 'Internal Server Error' }, { status: 500 })
  }
}
```

```ts
'use client'

import { useMutation, useQuery, useQueryClient } from '@tanstack/react-query'
import { memoriesClient, type MemoryRow } from '@/lib/api-client'
import type { MemoryCategory } from '@prisma/client'

export type MemoryScopeKind = 'project' | 'task'

interface UseMemoriesOptions {
  scope: MemoryScopeKind
  parentId: string
  enabled?: boolean
}

interface CreateInput {
  category: MemoryCategory
  content: unknown
}

/**
 * List + mutate memories for one task or one project. Mutations invalidate the
 * list query (no optimistic update — the tab is off the main path). Promote is
 * only meaningful for task scope; calling it on a project list is a no-op the
 * server rejects with 409.
 */
export function useMemories({ scope, parentId, enabled = true }: UseMemoriesOptions) {
  const queryClient = useQueryClient()
  const queryKey = ['memories', scope, parentId] as const

  const query = useQuery<{ memories: MemoryRow[] }>({
    queryKey,
    queryFn: () =>
      scope === 'project'
        ? memoriesClient.listForProject(parentId)
        : memoriesClient.listForTask(parentId),
    enabled: !!parentId && enabled,
  })

  const invalidate = () => queryClient.invalidateQueries({ queryKey })

  const createMutation = useMutation({
    mutationFn: (input: CreateInput) =>
      scope === 'project'
        ? memoriesClient.createForProject(parentId, input)
        : memoriesClient.createForTask(parentId, input),
    onSuccess: invalidate,
  })

  const editMutation = useMutation({
    mutationFn: ({ id, content }: { id: string; content: unknown }) =>
      memoriesClient.editContent(id, content),
    onSuccess: invalidate,
  })

  const promoteMutation = useMutation({
    mutationFn: (id: string) => memoriesClient.promoteToProject(id),
    onSuccess: () => {
      // Promote moves a row from task→project scope; refresh BOTH lists.
      invalidate()
      queryClient.invalidateQueries({ queryKey: ['memories', 'project'] })
    },
  })

  const removeMutation = useMutation({
    mutationFn: (id: string) => memoriesClient.remove(id),
    onSuccess: invalidate,
  })

  return {
    memories: query.data?.memories ?? [],
    loading: query.isPending && !!parentId && enabled,
    error: query.error ? ((query.error as Error).message ?? 'Failed to load') : null,
    refetch: query.refetch,
    create: (input: CreateInput) => createMutation.mutateAsync(input),
    edit: (id: string, content: unknown) => editMutation.mutateAsync({ id, content }),
    promote: (id: string) => promoteMutation.mutateAsync(id),
    remove: (id: string) => removeMutation.mutateAsync(id),
    mutating:
      createMutation.isPending ||
      editMutation.isPending ||
      promoteMutation.isPending ||
      removeMutation.isPending,
  }
}
```

```tsx
  | { category: 'CONVENTION'; value: ConventionValue }
  | { category: 'PROCEDURAL'; value: ProceduralValue }

function initialState(row?: MemoryRow): FormState | null {
  if (!row) return null
  switch (row.category) {
    case 'TRAP':
      return { category: 'TRAP', value: trapFromContent(row.content) }
    case 'DECISION':
      return { category: 'DECISION', value: decisionFromContent(row.content) }
    case 'CONVENTION':
      return { category: 'CONVENTION', value: conventionFromContent(row.content) }
    case 'PROCEDURAL':
      return { category: 'PROCEDURAL', value: proceduralFromContent(row.content) }
  }

function blankState(category: MemoryCategory): FormState {
  switch (category) {
    case 'TRAP':
      return { category: 'TRAP', value: trapDefault() }
    case 'DECISION':
      return { category: 'DECISION', value: decisionDefault() }
    case 'CONVENTION':
      return { category: 'CONVENTION', value: conventionDefault() }
    case 'PROCEDURAL':
      return { category: 'PROCEDURAL', value: proceduralDefault() }
  }
}

  switch (s.category) {
    case 'TRAP':
      return trapIsValid(s.value)
    case 'DECISION':
      return decisionIsValid(s.value)
    case 'CONVENTION':
      return conventionIsValid(s.value)
    case 'PROCEDURAL':
      return proceduralIsValid(s.value)
  }
}

function toContent(s: FormState): unknown {
  switch (s.category) {
    case 'TRAP':
      return trapToContent(s.value)
    case 'DECISION':
      return decisionToContent(s.value)
    case 'CONVENTION':
      return conventionToContent(s.value)
    case 'PROCEDURAL':
      return proceduralToContent(s.value)
```

```tsx
'use client'

import { useState } from 'react'
import { useIntl } from 'react-intl'
import { Brain, Plus } from 'lucide-react'
import type { MemoryRow } from '@/lib/api-client'
import {
  useMemories,
  type MemoryScopeKind,
} from '../../hooks/useMemories'
import { Button } from '@/components/ui/button'
import { MemoryListItem } from './MemoryListItem'
import {
  MemoryEditDialog,
  type MemorySubmitPayload,
} from './MemoryEditDialog'

interface MemoryTabProps {
  scope: MemoryScopeKind
  parentId: string
  enabled: boolean
  /** clerkId/memberId → display name, for "Added by X". Optional. */
  memberNames?: Map<string, string>
}

type DialogState =
  | { open: false }
  | { open: true; mode: 'create' }
  | { open: true; mode: 'edit'; editing: MemoryRow }

export function MemoryTab({
  scope,
  parentId,
  enabled,
  memberNames,
}: MemoryTabProps) {
  const intl = useIntl()
  const { memories, loading, error, create, edit, promote, remove, mutating } =
    useMemories({ scope, parentId, enabled })
  const [dialog, setDialog] = useState<DialogState>({ open: false })

  const handleSubmit = async (payload: MemorySubmitPayload) => {
    if (dialog.open && dialog.mode === 'edit') {
      await edit(dialog.editing.id, payload.content)
    } else {
      await create(payload)
    }
    setDialog({ open: false })
  }

  if (loading) {
    return (
      <div className='py-8 text-center text-xs text-zinc-400'>
        {intl.formatMessage({ id: 'agentRun.loading', defaultMessage: '加载中…' })}
      </div>
    )
  }

  if (error) {
    return <div className='py-8 text-center text-xs text-red-500'>{error}</div>
  }

  return (
    <div className='space-y-3'>
      <div className='flex items-center justify-between'>
        <span className='text-xs text-zinc-400'>
          {intl.formatMessage(
            { id: 'memory.tab.count', defaultMessage: '{count} memories' },
            { count: memories.length }
          )}
        </span>
        <Button
          size='sm'
          variant='outline'
          onClick={() => setDialog({ open: true, mode: 'create' })}
        >
          <Plus className='size-3.5 mr-1' />
          {intl.formatMessage({ id: 'memory.action.add' })}
        </Button>
      </div>

      {memories.length === 0 ? (
        <div className='py-12 flex flex-col items-center gap-2 text-zinc-400 dark:text-zinc-500'>
          <Brain className='size-6' />
          <p className='text-sm font-medium'>
            {intl.formatMessage({ id: 'memory.tab.empty.title' })}
          </p>
          <p className='text-[11px] max-w-xs text-center'>
            {intl.formatMessage({ id: 'memory.tab.empty.description' })}
          </p>
          <Button
            size='sm'
            variant='ghost'
            onClick={() => setDialog({ open: true, mode: 'create' })}
          >
            {intl.formatMessage({ id: 'memory.tab.empty.cta' })}
          </Button>
        </div>
      ) : (
        <div className='space-y-2'>
          {memories.map(m => (
            <MemoryListItem
              key={m.id}
              memory={m}
              canPromote={scope === 'task' && m.scope === 'TASK'}
              memberName={
                m.createdByMemberId
                  ? memberNames?.get(m.createdByMemberId)
                  : undefined
              }
              onEdit={editing => setDialog({ open: true, mode: 'edit', editing })}
              onPromote={mem => promote(mem.id)}
              onDelete={mem => remove(mem.id)}
            />
          ))}
        </div>
      )}

      {dialog.open && (
        <MemoryEditDialog
          // Remount per open so internal form state resets cleanly.
          key={dialog.mode === 'edit' ? dialog.editing.id : 'create'}
          open
          mode={dialog.mode}
          editing={dialog.mode === 'edit' ? dialog.editing : undefined}
          onOpenChange={open => {
            if (!open) setDialog({ open: false })
          }}
          onSubmit={handleSubmit}
          submitting={mutating}
        />
      )}
    </div>
  )
}
```
