# Lumo Remote MCP Server Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the local stdio MCP server (`mcp/server.ts`) with a public HTTP MCP server at `/api/mcp` that uses Clerk as the OAuth 2.1 authorization server, binds each MCP connection to one Lumo workspace, and exposes ~29 read + ~23 write tools across the Lumo domain.

**Architecture:** Next.js app hosts `/api/mcp` (streamable HTTP). `@clerk/mcp-tools` + `mcp-handler` handle OAuth and MCP protocol. A new `McpConnection` table tracks each user/client binding; `McpAuditLog` records every tool call. Tool handlers are transport-agnostic and call existing `lib/services/*` — no Prisma access in the tool layer.

**Tech Stack:** Next.js 14 App Router, TypeScript strict, Prisma 7 + Neon Postgres, Clerk, `@clerk/mcp-tools`, `mcp-handler`, `@modelcontextprotocol/sdk`, `@upstash/ratelimit`, `@vercel/kv`, Jest, Zod.

**Spec:** [`docs/superpowers/specs/2026-04-16-lumo-remote-mcp-server-design.md`](../specs/2026-04-16-lumo-remote-mcp-server-design.md)

import { prisma } from '@/lib/db/prisma'

export interface UpsertOnConnectInput {
  clerkUserId: string
  clientId: string
  clientName: string | null
}

export const mcpConnectionRepository = {
  findByClerkUserAndClient(clerkUserId: string, clientId: string) {
    return prisma.mcpConnection.findUnique({
      where: { clerkUserId_clientId: { clerkUserId, clientId } },
    })
  },

  findById(id: string) {
    return prisma.mcpConnection.findUnique({ where: { id } })
  },

  listActiveByClerkUser(clerkUserId: string) {
    return prisma.mcpConnection.findMany({
      where: { clerkUserId, revokedAt: null },
      orderBy: { lastUsedAt: 'desc' },
    })
  },

  upsertOnConnect(input: UpsertOnConnectInput) {
    return prisma.mcpConnection.upsert({
      where: {
        clerkUserId_clientId: {
          clerkUserId: input.clerkUserId,
          clientId: input.clientId,
        },
      },
      create: {
        clerkUserId: input.clerkUserId,
        clientId: input.clientId,
        clientName: input.clientName,
      },
      update: {
        clientName: input.clientName,
        lastUsedAt: new Date(),
        revokedAt: null,
      },
    })
  },

  setActiveWorkspace(id: string, workspaceId: string, opts: { auto: boolean }) {
    return prisma.mcpConnection.update({
      where: { id },
      data: {
        activeWorkspaceId: workspaceId,
        autoBoundAt: opts.auto ? new Date() : null,
      },
    })
  },

  touchLastUsed(id: string) {
    return prisma.mcpConnection.update({
      where: { id },
      data: { lastUsedAt: new Date() },
    })
  },

  revoke(id: string) {
    return prisma.mcpConnection.update({
      where: { id },
      data: { revokedAt: new Date(), activeWorkspaceId: null },
    })
  },
}
```

import { mcpConnectionRepository } from '@/lib/repositories/mcp-connection.repository'
import { memberRepository } from '@/lib/repositories/member.repository'

export interface RegisterOrTouchInput {
  clerkUserId: string
  clientId: string
  clientName: string | null
}

function httpError(status: number, message: string) {
  return Object.assign(new Error(message), { status })
}

export const mcpConnectionService = {
  registerOrTouch(input: RegisterOrTouchInput) {
    return mcpConnectionRepository.upsertOnConnect(input)
  },

  async bindWorkspace(
    connectionId: string,
    clerkUserId: string,
    workspaceId: string,
    opts: { auto: boolean }
  ) {
    const memberships = await memberRepository.findByClerkId(clerkUserId)
    const match = memberships.find(m => m.workspaceId === workspaceId)
    if (!match) {
      throw httpError(403, `Not a member of workspace ${workspaceId}`)
    }
    return mcpConnectionRepository.setActiveWorkspace(
      connectionId,
      workspaceId,
      opts
    )
  },

  listConnections(clerkUserId: string) {
    return mcpConnectionRepository.listActiveByClerkUser(clerkUserId)
  },

  async disconnect(connectionId: string, clerkUserId: string) {
    const existing = await mcpConnectionRepository.findById(connectionId)
    if (!existing || existing.clerkUserId !== clerkUserId) {
      throw httpError(404, `Connection ${connectionId} not found`)
    }
    return mcpConnectionRepository.revoke(connectionId)
  },
}
```

```typescript
import { ZodError } from 'zod'

export const McpErrorCode = {
  ResourceNotFound: -32001,
  WorkspaceRequired: -32002,
  RateLimit: -32005,
  InvalidParams: -32602,
  Forbidden: -32603,
  Internal: -32603,
} as const

export class McpError extends Error {
  constructor(
    public readonly code: number,
    message: string,
    public readonly data?: unknown
  ) {
    super(message)
    this.name = 'McpError'
  }
}

export function toMcpError(err: unknown): McpError {
  if (err instanceof McpError) return err
  if (err instanceof ZodError) {
    return new McpError(McpErrorCode.InvalidParams, 'Invalid parameters', {
      issues: err.issues,
    })
  }
  if (err && typeof err === 'object') {
    const maybe = err as { code?: string; status?: number; message?: string }
    if (maybe.code === 'RATE_LIMIT') {
      return new McpError(
        McpErrorCode.RateLimit,
        maybe.message ?? 'Too Many Requests'
      )
    }
    if (maybe.code === 'WORKSPACE_REQUIRED') {
      return new McpError(
        McpErrorCode.WorkspaceRequired,
        maybe.message ?? 'Workspace required — call select_workspace first'
      )
    }
    if (maybe.status === 404) {
      return new McpError(
        McpErrorCode.ResourceNotFound,
        maybe.message ?? 'Resource not found'
      )
    }
    if (maybe.status === 403) {
      return new McpError(McpErrorCode.Forbidden, maybe.message ?? 'Forbidden')
    }
  }
  const message = err instanceof Error ? err.message : String(err)
  return new McpError(McpErrorCode.Internal, message)
}
```

export type WorkspaceRole = 'OWNER' | 'ADMIN' | 'MEMBER' | 'VIEWER'

export interface ToolContext {
  clerkUserId: string
  connectionId: string
  workspaceId?: string
  memberId?: string
  memberRole?: WorkspaceRole
}

export function requireRole(ctx: ToolContext, allowed: WorkspaceRole[]): void {
  if (!ctx.memberRole || !allowed.includes(ctx.memberRole)) {
    throw new McpError(
      McpErrorCode.Forbidden,
      `This tool requires one of: ${allowed.join(', ')}. Current role: ${ctx.memberRole ?? 'unknown'}.`

export function requireWorkspace(

    throw Object.assign(

      { code: 'WORKSPACE_REQUIRED' }
    )
  }
}

import { mcpConnectionRepository } from '@/lib/repositories/mcp-connection.repository'
import { memberRepository } from '@/lib/repositories/member.repository'
import type { ToolContext, WorkspaceRole } from './auth'

export interface ResolveLumoContextInput {
  clerkUserId: string
  clientId: string
  clientName: string | null
}

export async function resolveLumoContext(
  input: ResolveLumoContextInput
): Promise<ToolContext> {
  const connection = await mcpConnectionRepository.upsertOnConnect(input)
  const memberships = await memberRepository.findByClerkId(input.clerkUserId)

  const bound = connection.activeWorkspaceId
    ? memberships.find(m => m.workspaceId === connection.activeWorkspaceId)
    : undefined

  if (bound) {
    return buildCtx(input.clerkUserId, connection.id, bound)
  }

  if (memberships.length === 1) {
    await mcpConnectionRepository.setActiveWorkspace(
      connection.id,
      memberships[0].workspaceId,
      { auto: true }
    )
    return buildCtx(input.clerkUserId, connection.id, memberships[0])
  }

  return {
    clerkUserId: input.clerkUserId,
    connectionId: connection.id,
  }
}

function buildCtx(
  clerkUserId: string,
  connectionId: string,
  membership: { id: string; workspaceId: string; role: string }
): ToolContext {
  return {
    clerkUserId,
    connectionId,
    workspaceId: membership.workspaceId,
    memberId: membership.id,
    memberRole: membership.role as WorkspaceRole,
  }
}

import { Ratelimit } from '@upstash/ratelimit'
import { kv } from '@vercel/kv'

export const BOOTSTRAP_TOOLS = new Set([
  'list_my_workspaces',
  'select_workspace',
  'get_active_workspace',
])

const ratelimit = new Ratelimit({
  redis: kv,
  limiter: Ratelimit.slidingWindow(60, '60 s'),
  prefix: 'mcp:rate',
  analytics: false,
})

export async function checkRateLimit(input: {
  clerkUserId: string
  toolName: string
}): Promise<{ success: true; remaining?: number }> {
  if (BOOTSTRAP_TOOLS.has(input.toolName)) {
    return { success: true }
  }
  const result = await ratelimit.limit(input.clerkUserId)
  if (!result.success) {
    throw Object.assign(new Error('Rate limit exceeded (60 req/min)'), {
      code: 'RATE_LIMIT',
    })
  }
  return { success: true, remaining: result.remaining }
}

```typescript
import { McpServer } from '@modelcontextprotocol/sdk/server/mcp.js'
import { registerAllTools, type CtxProvider } from './tools'
import { checkRateLimit } from './rate-limit'
import { writeAuditLog } from './audit'
import { toMcpError } from './errors'

export function buildMcpServer(ctxProvider: CtxProvider): McpServer {
  const server = new McpServer({ name: 'lumo', version: '1.0.0' })

  const originalRegister = server.registerTool.bind(server)
  ;(server as any).registerTool = (name: string, def: any, handler: any) => {
    const wrapped = async (args: any) => {
      const ctx = await ctxProvider()
      const started = Date.now()
      try {
        await checkRateLimit({ clerkUserId: ctx.clerkUserId, toolName: name })
        const result = await handler(args)
        await writeAuditLog({
          connectionId: ctx.connectionId,
          clerkUserId: ctx.clerkUserId,
          workspaceId: ctx.workspaceId ?? null,
          toolName: name,
          status: 'ok',
          durationMs: Date.now() - started,
        })
        return result
      } catch (err) {
        const mcpErr = toMcpError(err)
        await writeAuditLog({
          connectionId: ctx.connectionId,
          clerkUserId: ctx.clerkUserId,
          workspaceId: ctx.workspaceId ?? null,
          toolName: name,
          status: 'error',
          errorCode: String(mcpErr.code),
          durationMs: Date.now() - started,
        })
        throw mcpErr
      }
    }
    return originalRegister(name, def, wrapped)
  }

  registerAllTools(server, ctxProvider)
  return server
}
```

import { z } from 'zod'
import type { McpServer } from '@modelcontextprotocol/sdk/server/mcp.js'
import type { ToolContext } from '@/lib/mcp/auth'
import { requireWorkspace } from '@/lib/mcp/auth'
import { projectService } from '@/lib/services/project.service'

type CtxProvider = () => Promise<ToolContext>

function httpError(status: number, message: string) {
  return Object.assign(new Error(message), { status })
}

export function makeProjectHandlers(ctxProvider: CtxProvider) {
  return {
    async list_projects(_args: Record<string, never>) {
      const ctx = await ctxProvider()
      requireWorkspace(ctx)
      const projects = await projectService.listByWorkspace(ctx.workspaceId)
      return { projects }
    },

    async get_project(args: { projectId: string }) {
      const ctx = await ctxProvider()
      requireWorkspace(ctx)
      const project = await projectService.findById(args.projectId)
      if (!project || project.workspaceId !== ctx.workspaceId) {
        throw httpError(404, `Project ${args.projectId} not found`)
      }
      return { project }
    },
  }
}

export function registerProjectTools(
  server: McpServer,
  ctxProvider: CtxProvider
) {
  const h = makeProjectHandlers(ctxProvider)

  server.registerTool(
    'list_projects',
    {
      description: 'List all projects in the active workspace.',
      inputSchema: {},
    },
    async () => {
      const result = await h.list_projects({})
      return {
        content: [
          { type: 'text' as const, text: JSON.stringify(result, null, 2) },
        ],
      }
    }
  )

  server.registerTool(
    'get_project',
    {
      description: 'Get a project by ID.',
      inputSchema: { projectId: z.string() },
    },
    async args => {
      const result = await h.get_project(args)
      return {
        content: [
          { type: 'text' as const, text: JSON.stringify(result, null, 2) },
        ],
      }
    }
  )
}

import { NextResponse } from 'next/server'
import { auth } from '@clerk/nextjs/server'
import { mcpConnectionService } from '@/lib/services/mcp-connection.service'

export async function GET() {
  const { userId } = await auth()
  if (!userId) return new NextResponse('Unauthorized', { status: 401 })
  const connections = await mcpConnectionService.listConnections(userId)
  return NextResponse.json({ connections })
}

export async function DELETE(req: Request) {
  const { userId } = await auth()
  if (!userId) return new NextResponse('Unauthorized', { status: 401 })
  const { id } = (await req.json()) as { id: string }
  await mcpConnectionService.disconnect(id, userId)
  return NextResponse.json({ ok: true })
}
