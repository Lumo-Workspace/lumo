# @lumo/cli Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build `@lumo/cli`, an npx-runnable CLI tool for login authentication and viewing assigned tasks.

**Architecture:** New `cli/` workspace in the monorepo. Commander.js for arg parsing, chalk for colors, tsup to bundle a single CJS file. The CLI makes HTTP calls to the Lumo API with `Authorization: Bearer lum_xxx`. Backend adds a `requireAuthAny()` dual-auth function (Clerk or API Key) applied only to `/api/tasks/me` and `/api/workspaces/me`.

**Tech Stack:** TypeScript, Commander.js, chalk, tsup, Node.js built-in `crypto` and `fs`, Jest for tests.

import * as fs from 'fs'
import * as path from 'path'
import * as os from 'os'
import type { LumoConfig } from '../types'

function configPath(homeDir?: string): string {
  const home = homeDir ?? os.homedir()
  return path.join(home, '.lumo', 'config.json')
}

/**
 * Read the Lumo config file. Returns null if it doesn't exist or is invalid.
 * @param homeDir - override home directory (for testing)
 */
export function readConfig(homeDir?: string): LumoConfig | null {
  const filePath = configPath(homeDir)
  try {
    const raw = fs.readFileSync(filePath, 'utf-8')
    return JSON.parse(raw) as LumoConfig
  } catch {
    return null
  }
}

/**
 * Write the Lumo config file. Creates ~/.lumo/ if needed.
 * @param config - the config to write
 * @param homeDir - override home directory (for testing)
 */
export function writeConfig(config: LumoConfig, homeDir?: string): void {
  const filePath = configPath(homeDir)
  const dir = path.dirname(filePath)
  fs.mkdirSync(dir, { recursive: true })
  fs.writeFileSync(filePath, JSON.stringify(config, null, 2), {
    mode: 0o600,
  })
}

/**
 * Delete the Lumo config file. No-op if it doesn't exist.
 * @param homeDir - override home directory (for testing)
 */
export function deleteConfig(homeDir?: string): void {
  const filePath = configPath(homeDir)
  try {
    fs.unlinkSync(filePath)
  } catch {
    // File doesn't exist — that's fine
  }
}
```

import type { LumoConfig } from '../types'
import { LumoApiError } from '../types'

interface LumoClient {
  get<T>(path: string): Promise<T>
  post<T>(path: string, body?: unknown): Promise<T>
}

export function createClient(config: LumoConfig): LumoClient {
  const { apiKey, apiUrl } = config

  const headers: Record<string, string> = {
    Authorization: `Bearer ${apiKey}`,
    'Content-Type': 'application/json',
  }

  async function request<T>(
    method: string,
    path: string,
    body?: unknown
  ): Promise<T> {
    let response: Response
    try {
      response = await fetch(`${apiUrl}${path}`, {
        method,
        headers,
        ...(body !== undefined ? { body: JSON.stringify(body) } : {}),
      })
    } catch {
      throw new LumoApiError(
        0,
        `Cannot reach Lumo API at ${apiUrl}. Check your connection.`
      )
    }

    if (!response.ok) {
      let message: string
      try {
        const data = (await response.json()) as { error?: string }
        message = data.error ?? `${response.status} error`
      } catch {
        message = `${response.status} error`
      }
      throw new LumoApiError(response.status, message)
    }

    return response.json() as Promise<T>
  }

  return {
    get<T>(path: string): Promise<T> {
      return request<T>('GET', path)
    },
    post<T>(path: string, body?: unknown): Promise<T> {
      return request<T>('POST', path, body)
    },
  }
}
```

import chalk from 'chalk'

export interface TaskRow {
  identifier: string
  title: string
  status: string
  priority: string
  assigneeName: string
}

const STATUS_COLOR: Record<string, (s: string) => string> = {
  TODO: chalk.gray,
  IN_PROGRESS: chalk.blue,
  IN_REVIEW: chalk.magenta,
  DONE: chalk.green,
}

const PRIORITY_COLOR: Record<string, (s: string) => string> = {
  URGENT: chalk.red,
  HIGH: chalk.yellow,
  MEDIUM: chalk.cyan,
  LOW: chalk.gray,
}

function colorStatus(status: string): string {
  const colorFn = STATUS_COLOR[status]
  return colorFn ? colorFn(status) : status
}

function colorPriority(priority: string): string {
  const colorFn = PRIORITY_COLOR[priority]
  return colorFn ? colorFn(priority) : priority
}

/**
 * Format tasks into an aligned table string.
 */
export function formatTaskTable(tasks: TaskRow[]): string {
  if (tasks.length === 0) {
    return chalk.gray('  No tasks found.')
  }

  // Calculate column widths
  const idWidth = Math.max(...tasks.map(t => t.identifier.length))
  const titleWidth = Math.max(...tasks.map(t => t.title.length), 5)
  const statusWidth = Math.max(...tasks.map(t => t.status.length))
  const priorityWidth = Math.max(...tasks.map(t => t.priority.length))

  const lines = tasks.map(t => {
    const id = t.identifier.padEnd(idWidth)
    const title = t.title.padEnd(titleWidth)
    const status = t.status.padEnd(statusWidth)
    const priority = t.priority.padEnd(priorityWidth)

    return `  ${chalk.bold(id)}  ${title}  ${colorStatus(status)}  ${colorPriority(priority)}  ${chalk.dim('\u2192')} ${t.assigneeName}`
  })

  return lines.join('\n')
}

/**
 * Format the task count summary line.
 */
export function formatStatusCount(count: number): string {
  const noun = count === 1 ? 'task' : 'tasks'
  return chalk.dim(`\n  ${count} ${noun} assigned to you`)
}
```

import { createHash } from 'crypto'
import { prisma } from '@/lib/db/prisma'

interface ApiKeyAuthResult {
  userId: string
  workspaceId: string
  memberId: string
}

/**
 * Authenticate an API Key from the Authorization header.
 * Expects format: "Bearer lum_xxx..."
 *
 * Returns auth context on success, null on failure.
 */
export async function authenticateApiKey(
  headerValue: string
): Promise<ApiKeyAuthResult | null> {
  // 1. Extract key from "Bearer lum_xxx" format
  if (!headerValue.startsWith('Bearer ')) {
    return null
  }
  const key = headerValue.slice(7) // Remove "Bearer "

  // 2. Validate prefix
  if (!key.startsWith('lum_')) {
    return null
  }

  // 3. SHA-256 hash the key
  const keyHash = createHash('sha256').update(key).digest('hex')

  // 4. Look up in database
  const apiKey = await prisma.apiKey.findUnique({
    where: { keyHash },
    select: {
      id: true,
      revokedAt: true,
      member: {
        select: {
          clerkId: true,
          workspaceId: true,
          id: true,
        },
      },
    },
  })

  if (!apiKey) {
    return null
  }

  // 5. Check not revoked
  if (apiKey.revokedAt !== null) {
    return null
  }

  // 6. Update lastUsedAt (fire-and-forget is fine, but we await for correctness)
  await prisma.apiKey.update({
    where: { id: apiKey.id },
    data: { lastUsedAt: new Date() },
  })

  // 7. Return auth context matching requireAuth() shape
  return {
    userId: apiKey.member.clerkId,
    workspaceId: apiKey.member.workspaceId,
    memberId: apiKey.member.id,
  }
}

/**
 * Dual authentication: supports both Clerk (web) and API Key (CLI).
 * Checks Authorization header for "Bearer lum_..." first.
 * Falls back to Clerk session auth if no API Key header.
 * Returns { userId: string } on success, or a 401 NextResponse.
 */
export async function requireAuthAny(): Promise<
  { userId: string } | NextResponse
> {
  const headerStore = await headers()
  const authHeader = headerStore.get('authorization')

  // If Authorization header contains a Lumo API key, use API Key auth
  if (authHeader && authHeader.startsWith('Bearer lum_')) {
    const result = await authenticateApiKey(authHeader)
    if (!result) {
      return NextResponse.json(
        { error: 'API key invalid or revoked' },
        { status: 401 }
      )
    }
    return { userId: result.userId }
  }

  // Otherwise, fall back to Clerk session auth
  const { userId } = await auth()
  if (!userId) {
    return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })
  }
  return { userId }
}

import chalk from 'chalk'
import { writeConfig } from '../lib/config'

interface LoginOptions {
  key: string
  apiUrl: string
}

interface WorkspaceMeResponse {
  workspace: { name: string }
  role: string
}

/**
 * Handle the `lumo login` command.
 * Validates the API key, calls /api/workspaces/me, and saves config.
 */
export async function handleLogin(options: LoginOptions): Promise<void> {
  const { key, apiUrl } = options

  // Validate key format
  if (!key.startsWith('lum_')) {
    process.stderr.write(
      chalk.red(
        'Error: API key must start with "lum_". Check your key and try again.\n'
      )
    )
    process.exit(1)
  }

  // Validate against API
  let data: WorkspaceMeResponse
  try {
    const response = await fetch(`${apiUrl}/api/workspaces/me`, {
      method: 'GET',
      headers: {
        Authorization: `Bearer ${key}`,
        'Content-Type': 'application/json',
      },
    })

    if (!response.ok) {
      if (response.status === 401) {
        process.stderr.write(
          chalk.red(
            'Error: API key invalid or revoked. Check your key and try again.\n'
          )
        )
        process.exit(1)
      }
      const body = (await response
        .json()
        .catch(() => ({ error: 'Unknown error' }))) as { error?: string }
      process.stderr.write(
        chalk.red(
          `Error: ${response.status} ${body.error ?? 'Unknown error'}\n`
        )
      )
      process.exit(1)
    }

    data = (await response.json()) as WorkspaceMeResponse
  } catch (error) {
    if (error instanceof Error && error.message === 'process.exit called') {
      throw error // Re-throw test mock
    }
    process.stderr.write(
      chalk.red(
        `Error: Cannot reach Lumo API at ${apiUrl}. Check your connection.\n`
      )
    )
    process.exit(1)
  }

  // Save config
  writeConfig({
    apiKey: key,
    apiUrl,
    workspaceName: data.workspace.name,
    userName: '',
  })

  process.stdout.write(
    `\n${chalk.green('\u2713')} Logged in (workspace: ${chalk.bold(data.workspace.name)})\n` +
      chalk.dim(`  Config saved to ~/.lumo/config.json\n`)
  )
}

import chalk from 'chalk'
import { readConfig } from '../lib/config'
import { createClient } from '../lib/client'
import { formatTaskTable, formatStatusCount } from '../lib/output'
import type { TaskRow } from '../lib/output'

interface TasksOptions {
  all?: boolean
  status?: string
  project?: string
  json?: boolean
}

/** Priority sort order: URGENT first, LOW last */
const PRIORITY_ORDER: Record<string, number> = {
  URGENT: 0,
  HIGH: 1,
  MEDIUM: 2,
  LOW: 3,
}

interface ApiTask {
  id: string
  number: number
  title: string
  status: string
  priority: string
  teamIdentifier: string
  assignees?: Array<{ name: string }>
}

interface TasksMeResponse {
  tasks: ApiTask[]
  myTasksViewId: string | null
}

/**
 * Handle the `lumo tasks` command.
 * Lists tasks assigned to the current user.
 */
export async function handleTasks(options: TasksOptions): Promise<void> {
  // Read config
  const cfg = readConfig()
  if (!cfg) {
    process.stderr.write(
      chalk.red('Error: Not logged in. Run "lumo login" first.\n')
    )
    process.exit(1)
  }

  // Fetch tasks
  const client = createClient(cfg)
  let data: TasksMeResponse
  try {
    data = await client.get<TasksMeResponse>('/api/tasks/me')
  } catch (error) {
    if (error instanceof Error && error.message === 'process.exit called') {
      throw error
    }
    const status = (error as { status?: number }).status
    if (status === 401) {
      process.stderr.write(
        chalk.red(
          'Error: API key invalid or revoked. Run "lumo login" to re-authenticate.\n'
        )
      )
      process.exit(1)
    }
    process.stderr.write(chalk.red(`Error: ${(error as Error).message}\n`))
    process.exit(1)
  }

  let tasks = data.tasks

  // Filter: exclude DONE by default unless --all
  if (!options.all) {
    tasks = tasks.filter(t => t.status !== 'DONE')
  }

  // Filter: by status
  if (options.status) {
    const statusFilter = options.status.toUpperCase()
    tasks = tasks.filter(t => t.status === statusFilter)
  }

  // Sort by priority
  tasks.sort((a, b) => {
    const aPri = PRIORITY_ORDER[a.priority] ?? 99
    const bPri = PRIORITY_ORDER[b.priority] ?? 99
    return aPri - bPri
  })

  // JSON output mode
  if (options.json) {
    process.stdout.write(JSON.stringify({ tasks }, null, 2) + '\n')
    return
  }

  // Table output
  const rows: TaskRow[] = tasks.map(t => ({
    identifier: `${t.teamIdentifier}-${t.number}`,
    title: t.title,
    status: t.status,
    priority: t.priority,
    assigneeName: t.assignees?.[0]?.name ?? '',
  }))

  process.stdout.write('\n' + formatTaskTable(rows) + '\n')
  process.stdout.write(formatStatusCount(rows.length) + '\n')
}
```
